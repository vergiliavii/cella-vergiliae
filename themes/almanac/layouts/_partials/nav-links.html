"variant" "header" or "drawer"). */ -}} {{- $ctx := .ctx -}} {{- $variant := .variant -}} {{- $lang := $ctx.Site.Language -}} {{- $here := $ctx.Page.RelPermalink -}} {{- with $ctx.Site.Menus.main }} {{- /* A menu's addresses are under the site's path already.
